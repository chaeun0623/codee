#include #define TRIG_PIN 9 #define ECHO_PIN 10 #define BUTTON_PIN 2 #define DHT_PIN 4 #define DHT_TYPE DHT11 DHT dht(DHT_PIN, DHT_TYPE); int stage = 0; int crushCount = 0; bool lastButtonState = HIGH; unsigned long lastButtonTime = 0; unsigned long lastTempTime = 0; unsigned long lastDistanceTime = 0; float initialTemp = -100; float currentTemp = 0; float targetTemp = 0; bool pressTiming = false; unsigned long pressStartTime = 0; void setup() { Serial.begin(9600); pinMode(TRIG_PIN, OUTPUT); pinMode(ECHO_PIN, INPUT); pinMode(BUTTON_PIN, INPUT_PULLUP); dht.begin(); Serial.println("SYSTEM_READY"); Serial.println("STEP1_READY"); } void loop() { // STEP 1: 버튼 10회로 플라스틱 분쇄 bool currentButtonState = digitalRead(BUTTON_PIN); if (lastButtonState == HIGH && currentButtonState == LOW && millis() - lastButtonTime > 200) { lastButtonTime = millis(); if (stage == 0 && crushCount < 10) { crushCount++; Serial.println("CRUSH"); Serial.print("COUNT:"); Serial.println(crushCount); if (crushCount == 10) { stage = 1; initialTemp = -100; Serial.println("CRUSH_DONE"); Serial.println("STEP2_READY"); } } } lastButtonState = currentButtonState; // STEP 2: 기준 온도보다 1도 상승하면 용융 완료 if (millis() - lastTempTime >= 2000) { lastTempTime = millis(); float t = dht.readTemperature(); if (!isnan(t)) { currentTemp = t; Serial.print("TEMP:"); Serial.println(currentTemp, 1); if (stage == 1) { if (initialTemp == -100) { initialTemp = currentTemp; targetTemp = initialTemp + 1.0; Serial.print("BASE_TEMP:"); Serial.println(initialTemp, 1); Serial.print("TARGET_TEMP:"); Serial.println(targetTemp, 1); } else if (currentTemp >= targetTemp) { stage = 2; Serial.println("MELT_DONE"); Serial.println("STEP3_READY"); } } } } // STEP 3: 초음파 센서 10cm 이내에서 4초 연속 감지 if (stage == 2 && millis() - lastDistanceTime >= 100) { lastDistanceTime = millis(); digitalWrite(TRIG_PIN, LOW); delayMicroseconds(2); digitalWrite(TRIG_PIN, HIGH); delayMicroseconds(10); digitalWrite(TRIG_PIN, LOW); long duration = pulseIn(ECHO_PIN, HIGH, 30000); float distance = 999; if (duration > 0) { distance = duration * 0.0343 / 2; } if (distance <= 10) { if (!pressTiming) { pressTiming = true; pressStartTime = millis(); Serial.println("PRESS_START"); } if (millis() - pressStartTime >= 4000) { stage = 3; pressTiming = false; Serial.println("PROCESS_DONE"); Serial.println("ALL_COMPLETE"); } } else { if (pressTiming) { pressTiming = false; Serial.println("PRESS_CANCEL"); } } } }
